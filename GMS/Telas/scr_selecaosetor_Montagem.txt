                  { _agora: Now() },

                      },
                      UpdateContext({ctxFalhaSelM: false});
                      IfError(
                          Refresh(Producao_Montagem);
                          Refresh(Paradas_Montagem),
                          UpdateContext({ctxFalhaSelM: true})
                      );
                      Concurrent(
                          ClearCollect(
                              colAbertasSelM,
                              ForAll(
                                  Filter(Producao_Montagem, Status = "EmAndamento") As _r,
                                  {
                                      ID:       _r.ID,
                                      CelulaK:  Upper(Trim(_r.Celula)),
                                      PN:       _r.PN,
                                      OT:       _r.OT,
                                      QtdOT:    _r.QtdOT,
                                      InicioEm: _r.InicioEm
                                  }
                              )
                          ),
                          ClearCollect(
                              colTurnoSelM,
                              ForAll(
                                  Filter(Producao_Montagem, Data >= _data) As _r,
                                  {
                                      CelulaK:      Upper(Trim(_r.Celula)),
                                      Data:         _r.Data,
                                      Turno:        Trim(_r.Turno),
                                      Status:       _r.Status,

                                  }
                              )
                          ),
                          ClearCollect(
                              colParadasSelM,
                              ForAll(
                                  Filter(Paradas_Montagem, Data >= DateAdd(_data, -2, TimeUnit.Days)) As _p,
                                  {
                                      CelulaK:        Upper(Trim(_p.Celula)),
                                      Data:           _p.Data,
                                      Turno:          Trim(_p.Turno),
                                      Motivo:         _p.Motivo,
                                      TempoParadoMin: _p.TempoParadoMin,
                                      OPVinculada:    _p.OPVinculada
                                  }
                              )
                          )
                      );
                      ClearCollect(
                          colEstadoCelSelM,
                          ForAll(
                              fx_CelulasMontagem As _c,
                              With(
                                  {
                                      _ab: Sort(
                                          Filter(colAbertasSelM As _a, _a.CelulaK = Upper(_c.Celula)),
                                          InicioEm,
                                          SortOrder.Ascending
                                      ),
                                      _tu: Filter(
                                          colTurnoSelM As _t,
                                          _t.CelulaK = Upper(_c.Celula) && _t.Data = _data && _t.Turno = _turno
                                      ),
                                      _pa: Filter(
                                          colParadasSelM As _p,
                                          _p.CelulaK = Upper(_c.Celula) && _p.Data = _data && _p.Turno = _turno &&
                                          _p.Motivo <> fx_MotivoTempoExcedido
                                      )
                                  },
                                  With(
                                      { _f: First(_ab) },
                                      With(
                                          {
                                              _t: If(
                                                  IsBlank(_f.ID),
                                                  0,
                                                  Coalesce(LookUp(Tempos, Material = Upper(Trim(_f.PN))).TempoMinPc, 0)
                                              ),
                                              _chave: _f.PN & " (OT " & Text(_f.OT, "[$-en-US]0") & ")"
                                          },
                                          {
                                              Celula:     _c.Celula,
                                              Abertas:    CountRows(_ab),
                                              OT:         _f.OT,
                                              PN:         _f.PN,
                                              QtdOT:      _f.QtdOT,
                                              Inicio:     _f.InicioEm,
                                              MetaMin:    If(_t <= 0, -1, RoundDown(_f.QtdOT * _t, 0)),
                                              DescMin:    If(
                                                              IsBlank(_f.ID),
                                                              0,
                                                              Sum(
                                                                  Filter(
                                                                      colParadasSelM As _d,
                                                                      _d.CelulaK = Upper(_c.Celula) &&
                                                                      _d.OPVinculada = _chave &&
                                                                      _d.Motivo <> fx_MotivoTempoExcedido
                                                                  ),
                                                                  TempoParadoMin
                                                              )
                                                          ),
                                              Fechadas:   CountRows(Filter(_tu As _x, _x.Status = "Finalizado")),
                                              Pecas:      Sum(Filter(_tu As _x, _x.Status = "Finalizado"), QtdProduzida),
                                              ParadoMin:  Sum(_pa, TempoParadoMin),
                                              AtivaTurno: !IsEmpty(_tu) || !IsEmpty(_pa)
                                          }
                                      )
                                  )
                              )
                          )
                      );
                      UpdateContext({ ctxAgora: _agora, ctxDataSelM: _data, ctxTurnoSelM: _turno, ctxUltOkSelM: If(ctxFalhaSelM, ctxUltOkSelM, _agora) })
                  )
              )
