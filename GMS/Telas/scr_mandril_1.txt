          Properties:
            BorderColor: =Color.Transparent
            BorderThickness: =0
            Fill: =Color.Transparent
            Font: =fx_FonteFamilia
            Height: =1
            HoverFill: =Color.Transparent
            OnSelect: |-
              =Refresh(Eventos_Producao);
              Refresh(ficha_producao);
              Set(varDiaRefMand, fx_DataRefEm(Now()));
              ClearCollect(
                  colEvDiaMand,
                  Filter(Eventos_Producao, data_ref >= varDiaRefMand)
              );
              ClearCollect(
                  colEvMandFimMand,
                  Filter(colEvDiaMand, etapa.Value = "Mandril", tipo.Value = "Fim")
              );
              ClearCollect(
                  colEvMandParcMand,
                  Filter(colEvDiaMand, etapa.Value = "Mandril", tipo.Value = "Parcial")
              );
              ClearCollect(colEvTurnoMand, colEvMandFimMand);
              Collect(colEvTurnoMand, colEvMandParcMand);
              ClearCollect(
                  colFilaTuboMand,
                  Filter(
                      ficha_producao,
                      processo = "Tubo",
                      status_processo = "Aguardando"
                  )
              );
              Set(
                  varKanbanTuboMand,
                  CountRows(
                      Filter(colFilaTuboMand As _f, !IsBlank(_f.'Carretel Entrada'))
                  )
              );
              With(
                  {
                      _nivel: If(
                          varKanbanTuboMand >= fx_KanbanAlvo,   "OK",
                          varKanbanTuboMand >= fx_KanbanMinimo, "ATENCAO",
                          "CRITICO"
                      )
                  },
                  If(
                      _nivel = "CRITICO" && varKanbanNivelMand <> "CRITICO",
                      Notify(
                          "Kanban do Tubo crítico: " & Text(varKanbanTuboMand) &
                          " carretel(éis) aguardando (mínimo " & Text(fx_KanbanMinimo) & ").",
                          NotificationType.Warning,
                          15000
                      )
                  );
                  Set(varKanbanNivelMand, _nivel)
              )
            PressedFill: =Color.Transparent
            Text: =""
            Width: =1
            X: =1364
            Y: =764
