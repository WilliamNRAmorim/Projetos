//  Qtd_Turno2 dentro do 1º turno, o Hora-Hora abria o turno errado e o
//  fx_FaixaVencida dava as 9 janelas do 1º turno como vencidas de uma vez
//  (turnoLinha ≠ turno do relógio → true). Agora os três turnos ladrilham o
//  dia sem sobra: 320 + 517 = 837 · 837 + 563 = 1400 (23:20, com hora extra) ·
//  1340 + 420 = 1760 − 1440 = 320.

fx_MinutosDoDia(agora: DateTime): Number =
    Hour(agora) * 60 + Minute(agora);

fx_Turno(minutosDoDia: Number): Number =
    If(
        minutosDoDia >= 320 && minutosDoDia < 837,  1,
        minutosDoDia >= 837 && minutosDoDia < 1361, 2,
        3
    );

fx_TurnoEm(agora: DateTime): Number =
    fx_Turno(fx_MinutosDoDia(agora));

fx_TurnoLabelEm(agora: DateTime): Text =
    Text(fx_TurnoEm(agora)) & "º Turno";

// Antes das 05:20 o apontamento pertence ao dia anterior.
fx_DataRefEm(agora: DateTime): Date =
    DateAdd(
        DateValue(agora),
        If(fx_MinutosDoDia(agora) < 320, -1, 0),
        TimeUnit.Days
    );

fx_JanelaInicioEm(agora: DateTime): DateTime =
    DateAdd(fx_DataRefEm(agora), 320, TimeUnit.Minutes);

fx_JanelaFimEm(agora: DateTime): DateTime =
    DateAdd(fx_JanelaInicioEm(agora), 1, TimeUnit.Days);

// Cronômetro único do GMS, com teto visual de 24 h.
fx_Crono(inicio: DateTime, agora: DateTime): Text =
    If(
        IsBlank(inicio),
        "--:--",
        With(
            { _m: Max(0, DateDiff(inicio, agora, TimeUnit.Minutes)) },
            If(
                _m >= 1440,
                "+24 h",
                Text(Int(_m / 60), "00") & ":" & Text(Mod(_m, 60), "00")
            )
        )
    );


//  §5 · TEXTO E NÚMERO  ═══════════════════════════════════════════════
//  Duas leituras de número, NÃO intercambiáveis:
//  fx_Num = digitado pelo operador · fx_NumSpec = vindo de planilha.
//  fx_Num("0.375") = 375 (milhar) · fx_NumSpec("0.375") = 0,375 (polegada).

// Normaliza código de peça para comparação (sem espaço, caixa baixa, sem milhar).
fx_NormCodigo(codigo: Text): Text =
    With(
        { _t: Substitute(Trim(codigo), " ", "") },
        Lower(
            If(
                !IsBlank(_t) && IsNumeric(_t),
                Text(Value(_t), "[$-en-US]#"),
                _t
            )
        )
    );

// O texto digitado é número válido? Use antes de fx_Num em DisplayMode/BorderColor:
// fx_Num de lixo devolve Blank, e Blank em Coalesce vira 0 — que grava zero calado.
fx_EhNum(txt: Text): Boolean =
    IsMatch(Trim(txt), "\d+([.,]\d+)*");

// Número digitado por humano: ponto e vírgula têm o mesmo peso, independe da
// cultura do aparelho. Blank se não for número.
fx_Num(txt: Text): Number =
    IfError(
        With(
            { _t: Substitute(Substitute(Trim(txt), " ", ""), Char(160), "") },
            With(
                { _seps: MatchAll(_t, "[.,]") },
                If(
                    CountRows(_seps) = 0,
                    Value(_t, "en-US"),
                    With(
                        {
                            _pos: Last(_seps).StartMatch,
                            _n: CountRows(_seps)
                        },
                        If(
                            _n = 1 && Len(_t) - _pos = 3,
                            Value(Substitute(Substitute(_t, ".", ""), ",", ""), "en-US"),
                            Value(
                                Substitute(Substitute(Left(_t, _pos - 1), ".", ""), ",", "")
                                & "."
                                & Mid(_t, _pos + 1),
                                "en-US"
                            )
                        )
                    )
                )
            )
        ),
        Blank()
    );

// Valor de spec vindo de planilha (DM_*). Vazio, traço ou lixo viram 0, nunca erro.
// Quem chama passa Text(campo) — a coluna pode ser texto ou número.
fx_NumSpec(txt: Text): Number =
    IfError(
        Value(Substitute(Trim(txt), ",", "."), "en-US"),
        0
    );

// [v3.0] Velocidade (m/h) a partir do TEXTO de _13TEORICA_M_H (DM_ENGRENAGENS).
// Regra da v2.7.3 (MEL-112), À PROVA DE IDIOMA: tem vírgula → vírgula é decimal e
// ponto é milhar; sem vírgula → ponto é decimal; sempre lido em "en-US".
// "-" (Spiral) e vazio viram 0. NÃO é fx_NumSpec: fx_NumSpec lê "1.116,0" como erro.
// Chamadores: fx_VelocidadeTeorica (§11) e a carga local do Painel (colVelTrPnl).
fx_NumVelocidade(txt: Text): Number =
    With(
        { _t: Substitute(Substitute(txt, " ", ""), Char(160), "") },
        With(
            {
                _en: If(
                    !IsBlank(Find(",", _t)),
                    Substitute(Substitute(_t, ".", ""), ",", "."),
                    _t
                )
            },
            IfError(Coalesce(Value(_en, "en-US"), 0), 0)
        )
    );

// Faixa de especificação. Spec ausente (0) nunca acusa: poka-yoke não grita sem base.
fx_ForaDaFaixa(lido: Number, minimo: Number, maximo: Number): Boolean =
    lido > 0 && minimo > 0 && maximo > 0 && (lido < minimo || lido > maximo);

// [v3.0.7 · MEL-202] NORMA William 05/10/2026: nenhuma etapa envia OP adiante com metragem em branco ou 0,
// do Mandril em diante. Toda ação "Enviar" confere com esta UDF ANTES de gravar. Em branco também reprova:
// Coalesce(x, 0) na gravação transformava dado ausente em zero calado.
fx_MetragemOk(metros: Number): Boolean =
    !IsBlank(metros) && metros > 0;

// [v3.0.8 · MEL-206] Fita match-mate da Bandagem. Decisão William 05/10/2026: a fita CONTINUA bloqueando quando a mangueira
// leva fita e o código digitado não bate com o Delineamento; mas "-" no Delineamento quer dizer "não leva fita" e o operador
// NÃO precisa digitar nada. Três estados da spec: "-" dispensa · em branco = não cadastrada (exige algo digitado e avisa a
// supervisão) · código = tem que bater (fx_NormCodigo dos dois lados).
fx_FitaDispensada(spec: Text): Boolean =
    Trim(spec) = "-";

fx_FitaOk(spec: Text, lido: Text): Boolean =
    With(
        { _s: fx_NormCodigo(spec), _l: fx_NormCodigo(lido) },
        If(
            fx_FitaDispensada(spec), true,
            IsBlank(_s),             !IsBlank(_l),
            _l = _s
        )
    );

// [v3.0.8 · MEL-207] Código do composto da Cobertura: obrigatório e nunca só zeros ("0", "00", "0-0"). William 05/10/2026:
// "composto 0" é erro de digitação, não código. A comparação com o Delineamento (COVER_PN) entra quando ele mostrar a tabela;
// o código correto, pelo William, é no formato 500-01 (hoje digitam também 500.01 e 50001).
fx_CompostoValido(cod: Text): Boolean =
    !IsBlank(Trim(cod)) && !IsMatch(Trim(cod), "[0.\-_ ]*");

// [v3.0.8 · MEL-202] Margem aceitável para a metragem SUBIR de uma etapa para a seguinte (William, 05/10/2026:
// "pode subir, mas com margem para não virar várzea"). Provisória em 10%: nos últimos 60 dias, só 2% a 8% das OPs
// passaram de 10% (Cobertura→Bandagem 2,4% · Debandagem→Extração 2,7% · Trançagem→Cobertura 7,9%) e as altas
// absurdas são erro de digitação. Vira aviso (não bloqueia). William confirma o número com o delineamento.
fx_MargemMetragemPct = 10;

// A metragem informada passou da margem sobre a da etapa anterior? Etapa anterior em branco ou 0 nunca acusa
// (poka-yoke não grita sem base).
fx_MetragemSubiu(atual: Number, anterior: Number): Boolean =
    !IsBlank(anterior) && anterior > 0 && !IsBlank(atual) &&
    atual > anterior * (1 + fx_MargemMetragemPct / 100);

// Diâmetro digitado em polegada. Aceita vírgula ou ponto ("0,412" = "0.412") e também o milésimo digitado
// ("412" = 0,412): diâmetro de mangueira acima de 5 pol não existe nesta fábrica, então valor > 5 é milésimo.
// Histórico: a coluna Diametro_Trançagem guardou milésimos até meados de setembro (55 a 70% > 5) e polegada depois.
// Blank se não for número. NÃO usar fx_Num aqui (lê "0.500" como 500).
fx_DiametroPol(txt: Text): Number =
    With(
        { _v: fx_NumMedida(txt) },
        If(IsBlank(_v), Blank(), If(_v > 5, _v / 1000, _v))
    );

// Conferência do fio da bobina contra os fios do delineamento (1ª e 2ª trama).
// "SEM_FIO" quando não há o que comparar — bobina sem código ou mangueira sem spec.
// Sem esse terceiro estado, dado faltando vira "divergente" e o operador aprende a
// ignorar o alerta. Três chamadores: badge do seletor, bloco CONFERÊNCIA e alerta da ficha.
fx_ConfFio(fioBobina: Text, fioTrama1: Text, fioTrama2: Text): Text =
    With(
        {
            _f: fx_NormCodigo(fioBobina),
            _a: fx_NormCodigo(fioTrama1),
            _b: fx_NormCodigo(fioTrama2)
        },
        If(
            IsBlank(_f) || (IsBlank(_a) && IsBlank(_b)), "SEM_FIO",
            _f = _a || _f = _b,                          "OK",
            "DIVERGE"
        )
    );

// Número de um parâmetro de receita ("110'" → 110 · "152°C" → 152 · "45 min" → 45).
fx_NumeroReceita(txt: Text): Number =
    IfError(
        Value(
            Substitute(Substitute(Substitute(Substitute(Substitute(Substitute(
                Lower(Trim(txt)),
            "'", ""), "°", ""), "min", ""), "c", ""), "m", ""), " ", "")
        ),
        0
    );

fx_QtdBR(valor: Number): Text =
    Substitute(Text(valor, "[$-en-US]0.00"), ".", ",");

// Medida dimensional digitada pelo operador (polegada, 3 casas). Ponto e virgula
// sao o MESMO separador decimal: "0,254" = "0.254" = 0,254. Nunca milhar.
// NAO usar fx_Num aqui: fx_Num le "0.500" como 500 (milhar), regra correta para
// metragem e catastrofica para diametro. Foi essa troca que corrompeu 1.454
// medidas entre 20/08 e 28/08/2026.
fx_NumMedida(txt: Text): Number =
    IfError(
        With(
            { _t: Substitute(Substitute(Substitute(Trim(txt), " ", ""), Char(160), ""), ",", ".") },
            If(IsBlank(_t), Blank(), Value(_t, "en-US"))
        ),
        Blank()
    );

// Medida com 3 casas e virgula decimal, para o operador ler.
// fx_QtdBR tem 2 casas e mostra 0,254 como 0,25 - nao serve para diametro.
fx_MedidaBR(valor: Number): Text =
    Substitute(Text(valor, "[$-en-US]0.000"), ".", ",");

// Escapa texto que vai dentro de SVG/HTML gerado por fórmula.
fx_XmlSeguro(txt: Text): Text =
    Substitute(Substitute(Substitute(txt, "&", "&amp;"), "<", "&lt;"), ">", "&gt;");

//  §8 · CÓDIGO DE BARRAS  ═════════════════════════════════════════════
//  Code 128 Set B (C21 · C35). START = 104; com 103 toda minúscula corrompe.

// Os 95 caracteres do Code Set B: valor = posição - 1.
fx_C128Charset = " !""#$%&'()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\]^_`abcdefghijklmnopqrstuvwxyz{|}~";

// Padrões 0..105 concatenados, 11 módulos cada. O padrão de n é Mid(_, n*11+1, 11).
fx_C128Patterns = "11011001100110011011001100110011010010011000100100011001000100110010011001000100110001001000110010011001001000110010001001100010010010110011100100110111001001100111010111001100100111011001001110011011001110010110010111001100100111011011100100110011101001110110111011101001100111001011001110010011011101100100111001101001110011001011011011000110110001101100011011010100011000100010110001000100011010110001000100011010001000110001011010001000110001010001100010001010110111000101100011101000110111010111011000101110001101000111011011101110110110100011101100010111011011101000110111000101101110111011101011000111010001101110001011011101101000111011000101110001101011101111010110010000101111000101010100110000101000011001001011000010010000110100001011001000010011010110010000101100001001001101000010011000010100001101001000011001011000010010110010100001111011101011000010100100011110101010011110010010111100100100111101011110010010011110100100111100101111010010011110010100111100100101101101111011011110110111101101101010111100010100011110100010111101011110100010111100010111101010001111010001010111011110101111011101110101111011110101110110100001001101001000011010011100";

fx_C128Val(c: Text): Number =
    Coalesce(Find(c, fx_C128Charset), 1) - 1;

fx_C128Pat(n: Number): Text =
    If(
        n = 106,
        "1100011101011",                        // STOP (13 módulos)
        Mid(fx_C128Patterns, n * 11 + 1, 11)
    );

// START B + dados + checksum + STOP.
fx_C128Bits(txt: Text): Text =
    If(
        IsBlank(txt),
        "",
        With(
            { _n: Len(txt) },
            With(
                {
                    _chk: Mod(
                        104 + Sum(
                            ForAll(
                                Sequence(_n) As _i,
                                fx_C128Val(Mid(txt, _i.Value, 1)) * _i.Value
                            ),
                            Value
                        ),
                        103
                    )
                },
                fx_C128Pat(104)
                & Concat(
                    ForAll(
                        Sequence(_n) As _i,
                        fx_C128Pat(fx_C128Val(Mid(txt, _i.Value, 1)))
                    ),
                    Value
                )
                & fx_C128Pat(_chk)
                & fx_C128Pat(106)
            )
        )
    );

//  Code 39 (MEL-239) — o bipador da logística lê Code 39 (é o do BD Packing List antigo, `*GH781-8*`).
//  Aceita 0-9 A-Z - . espaço $ / + % ; o hífen do part number passa. 9 elementos por caractere
//  (barra/espaço alternados), `w` = largo (3 módulos). Validado com leitor ZXing em 100%, 83% e 60% da escala.
fx_C39Charset = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ-. *$/+%";

fx_C39Patterns = "nnnwwnwnnwnnwnnnnwnnwwnnnnwwnwwnnnnnnnnwwnnnwwnnwwnnnnnnwwwnnnnnnnwnnwnwwnnwnnwnnnnwwnnwnnwnnnnwnnwnnwnnwnnwwnwnnwnnnnnnnwwnnwwnnnwwnnnnnwnwwnnnnnnnnwwnwwnnnnwwnnnnwnnwwnnnnnnwwwnnwnnnnnnwwnnwnnnnwwwnwnnnnwnnnnnwnnwwwnnnwnnwnnnwnwnnwnnnnnnnwwwwnnnnnwwnnnwnnnwwnnnnnwnwwnwwnnnnnnwnwwnnnnnwwwwnnnnnnnwnnwnnnwwwnnwnnnnnwwnwnnnnnwnnnnwnwwwnnnnwnnnwwnnnwnnnwnnwnwnnnwnwnwnnnnwnwnnnwnnwnnnwnwnnnnwnwnwn";

fx_C39Pat(c: Text): Text =
    Mid(fx_C39Patterns, (Coalesce(Find(c, fx_C39Charset), 1) - 1) * 9 + 1, 9);

// Módulos 1/0 de "*TEXTO*": barra larga = 111, espaço largo = 000, 1 módulo de espaço entre caracteres.
fx_C39Bits(txt: Text): Text =
    Concat(
        ForAll(
            Sequence(Len(txt)) As _i,
            With(
                { _p: fx_C39Pat(Mid(txt, _i.Value, 1)) },
                Concat(
                    ForAll(
                        Sequence(9) As _j,
                        If(
                            Mod(_j.Value, 2) = 1,
                            If(Mid(_p, _j.Value, 1) = "w", "111", "1"),
                            If(Mid(_p, _j.Value, 1) = "w", "000", "0")
                        )
                    ),
                    Value
                ) & "0"
            )
        ),
        Value
    );

// Minuto do dia em que cada turno começa: 05:20 · 13:57 · 22:41 (William, 05/10/2026).

fx_InicioTurnoMin(turno: Number): Number =

// Minutos decorridos desde o início do turno. Mod resolve a virada de meia-noite

fx_MinNoTurno(agora: DateTime, turno: Number): Number =

// A janela já fechou? Turno futuro nunca vence (senão a tela abre gritando quando
// o operador consulta o turno seguinte); turno passado vence inteiro.
fx_FaixaVencida(fimOffset: Number, turnoLinha: Number, agora: DateTime): Boolean =
    With(
        { _ta: fx_TurnoEm(agora) },
        If(
            turnoLinha = _ta,                fx_MinNoTurno(agora, turnoLinha) >= fimOffset,
            turnoLinha = Mod(_ta, 3) + 1,    false,
            true
        )
    );

// Pendência da linha, em um lugar só.
// "" = nada a cobrar · SEM_APONTAMENTO = janela fechada e vazia ·
// SEM_MOTIVO = ficou abaixo da meta e não disse por quê.
// Janela aberta nunca acusa: cobrar hora que ainda não terminou ensina o operador
// a ignorar o alerta (mesma regra do terceiro estado do fx_ConfFio).
// ⚠ As telas NÃO chamam esta aqui direto — elas chamam a fx_PendenciaMaqHH abaixo.
//   Esta fica exposta para quem não tem o conceito de recurso ativo.
fx_PendenciaHH(vencida: Boolean, mangueira: Text, meta: Number, real: Number, motivo: Text): Text =
    If(
        !vencida,                                          "",
        IsBlank(Trim(mangueira)) && real <= 0,             "SEM_APONTAMENTO",
        meta > 0 && real < meta && IsBlank(Trim(motivo)),  "SEM_MOTIVO",
        ""
    );

// [v2.6 · MEL-111] A MESMA pendência, com o filtro de MÁQUINA ATIVA por cima.
// Um recurso só é cobrado se deu sinal de vida no turno:
//   A) alguma janela já apontada (metro ou mangueira), ou
//   B) OP com status_processo = "Em processo" endereçada a ele na ficha_producao.
// Sem sinal, a máquina aparece cinza, "sem produção no turno", fora do contador.
// Histórico (360 turnos): 7,6 trançadeiras apontam por turno de 11, e as 11 rodam
// em 6,1% dos turnos — cobrar as 11 dava ~38 janelas falsas por turno, e alerta
// que grita sem base ensina o operador a ignorar o alerta (C1 · C17 · C24).
// A regra nasceu na v0.5.2 dentro do código da tela e MORREU na reescrita da
// v0.5.3. Por isso ela mora aqui agora: bloco de tela é reescrito, fx_ não.
// Chamadores: cor/borda da linha, borda do Real, borda do Motivo, faixa de alerta
// (cor, visibilidade e texto), trava do Fechar turno, contador da célula e o card
// do Hora-Hora na ficha da Trançagem.
fx_PendenciaMaqHH(ativa: Boolean, vencida: Boolean, mangueira: Text, meta: Number, real: Number, motivo: Text): Text =
    If(
        !ativa,
        "",
        fx_PendenciaHH(vencida, mangueira, meta, real, motivo)
    );

fx_RendimentoHH(meta: Number, real: Number): Number =

// [v2.6 · MEL-111] Qual máquina a tela está mostrando: a da OP selecionada; sem OP,

// Chave canônica da mangueira para casar o que o operador escolhe com o que está

// Abertura e fechamento de UMA janela da grade.
// `fim` e `minutos` vêm da fx_GradeHoraHora; `dataRef` vem do fx_DataRefEm.
// Somar minutos à data resolve sozinho a virada de meia-noite do 3º turno:
// 1340 + 420 = 1760 min = 05:20 do dia seguinte. Nada de comparar data com

fx_JanelaHHIni(dataRef: Date, turno: Number, fim: Number, minutos: Number): DateTime =
    DateAdd(dataRef, fx_InicioTurnoMin(turno) + fim - minutos, TimeUnit.Minutes);

fx_JanelaHHFim(dataRef: Date, turno: Number, fim: Number): DateTime =
    DateAdd(dataRef, fx_InicioTurnoMin(turno) + fim, TimeUnit.Minutes);

// Quantos minutos um carretel passou DENTRO de uma janela. É a primitiva do
// rateio: o carretel entrega metros à janela na proporção do tempo que rodou
// dentro dela.
// Carretel que atravessa a virada de turno entra nas janelas dos DOIS turnos,
// cada uma com a sua fatia — e é assim que tem que ser.
// Blank de qualquer lado devolve 0: carretel sem carimbo não rateia nada e não
// inventa produção.
// ⚠ Max()/Min() de DateTime não têm assinatura garantida no Power Fx — por isso
//   o recorte é feito com If explícito.
fx_OverlapMin(janIni: DateTime, janFim: DateTime, runIni: DateTime, runFim: DateTime): Number =
    If(
        IsBlank(runIni) || IsBlank(runFim) || IsBlank(janIni) || IsBlank(janFim),
        0,
        With(
            {
                _i: If(runIni > janIni, runIni, janIni),
                _f: If(runFim < janFim, runFim, janFim)
            },
            If(_f <= _i, 0, DateDiff(_i, _f, TimeUnit.Minutes))
        )
    );

// Quanto de uma grandeza do carretel (metros produzidos, perda) cai numa janela.
// Proporcional ao TEMPO — é a melhor aproximação sem sensor. Quando o Factbird
// entrar (C20), a contagem por minuto substitui isto e o rateio deixa de
// existir; até lá, é rateio declarado, não medição disfarçada.
// duracao <= 0 devolve 0 em vez de estourar: carretel de duração zero é dado
// ruim, e dado ruim não vira divisão por zero na tela do operador.
fx_RateioCob(valor: Number, minutosNaJanela: Number, duracaoTotal: Number): Number =
    If(
        duracaoTotal <= 0 || minutosNaJanela <= 0,
        0,
        Coalesce(valor, 0) * minutosNaJanela / duracaoTotal
    );

// ── [v3.0 · Painel v0.10.0] AUTONOMIA DE TUBO DAS TRANÇADEIRAS (C10 · C23) ──

// Turno útil em minutos (7h20, William 28/09). É a régua da autonomia de tubo:
// < 1 turno vermelho · 1 a < 2 turnos laranja · >= 2 turnos verde.
// ⚠ Não é a janela do Hora-Hora (1º turno = 517 min). Recalibrar AQUI, nunca na tela.
fx_TurnoUtilMin = 440;

// Autonomia AGORA: a calculada na carga (minCarga, no instante sinc) menos o que o
// relógio da tela andou desde então. Recebe os instantes, não lê Now() (regra 2).
fx_AutonomiaAgora(minCarga: Number, sinc: DateTime, agora: DateTime): Number =
    Max(0, minCarga - Max(0, DateDiff(sinc, agora, TimeUnit.Minutes)));

// "5h40" · "12h05" · "+24 h". Formato diferente do fx_Crono ("05:40") de propósito:
// crono é tempo que PASSOU, autonomia é tempo que FALTA — não podem se confundir na TV.
// [v3.0.2] Zero vira "Conferir": pela conta o carretel montado já acabou e não há fila, mas
// a OP segue Em processo. "0h00" com a máquina girando lia como painel quebrado, e "fim?"
// (v3.0.1) não servia numa TV lida por gerente e diretor. O motivo vai no Onde olhar
// ("tubo esgotado pela conta" + janelas sem Hora-Hora).
fx_AutonomiaTexto(minutos: Number): Text =
    With(
        { _m: Round(Max(0, minutos), 0) },
        If(
            _m <= 0,
            "Conferir",
            _m >= 1440,
            "+24 h",
            Text(Int(_m / 60), "[$-en-US]0") & "h" & Text(Mod(_m, 60), "[$-en-US]00")
        )
    );

// [v3.0.2] Abaixo disto, a metragem de um carretel na fila é apontamento errado e entra no
// Onde olhar (ex.: carretel de 1 m na Spiral, 30/09). Vazio conta como 0. Recalibrar aqui.
fx_MetrosMinCarretel = 10;

// Real líquido de uma OT aberta, em minutos: bruto desde o início menos as paradas
// vinculadas a ela (não-TE). É a conta da scr_Montagem_HoraHora v0.5.0 (lbl_prow_real_m_1),
// agora num lugar só. Recebe o instante (regra 2 do cabeçalho). Sem início, 0.
fx_RealMontagemMin(inicio: DateTime, descontoMin: Number, agora: DateTime): Number =
    If(
        IsBlank(inicio),
        0,
        Max(0, DateDiff(inicio, agora, TimeUnit.Minutes) - Coalesce(descontoMin, 0))
    );

//  §16 · CONFERÊNCIA GMS × SAP × PACKING (C29 · MEL-196)  ═════════════
//  [v3.0.4] A conta é do fluxo GMS_Conferencia_Diaria (Office Script sobre o MB51); o app
//  só LÊ a lista Conferencia_SAP e grava a tratativa. Prioridade vem pronta do script:
//  1–7 e 10 = GMS × SAP por OP · 8–9 = refugo · 11–13 = packing · 99 = resolvido.

// Metros com milhar e 1 casa no padrão BR (1.138,5). Vazio mostra "—", nunca 0 (§2.13).
fx_MetrosBR(valor: Number): Text =
    If(
        IsBlank(valor),
        "—",
        With(
            { _t: Text(Round(valor, 1), "[$-en-US]#,##0.0") },
            Substitute(Substitute(Substitute(_t, ",", "#"), ".", ","), "#", ".")
        )
    );

// [v3.0.7 · MEL-201] Metros com unidade para exibição em card/label: vazio mostra "—", nunca "0 m".
// Substitui Coalesce(campo, 0) & " m", que mostrava metragem inexistente como zero (regra 6 do CLAUDE.md).
// Inteiro sai sem decimal ("1.138 m"); fracionado com 1 casa ("1.138,5 m"), milhar com ponto, decimal com vírgula.
fx_MetrosUn(valor: Number): Text =
    If(
        IsBlank(valor),
        "—",
        Substitute(
            Substitute(
                Substitute(
                    Text(valor, If(valor = Int(valor), "[$-en-US]#,##0", "[$-en-US]#,##0.0")),
                    ",", "#"
                ),
                ".", ","
            ),
            "#", "."
        ) & " m"
    );

// Filtro da tela: em que bloco a pendência cai.
