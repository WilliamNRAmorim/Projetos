              IfError(
                  ClearCollect(colHistMandrilDia, Filter(Eventos_Producao, data_ref = ctxHistData, etapa.Value = "Mandril"));

                  With(
                      {_turno: Switch(cmb_hist_turno_mandril.Selected.Value, "Turno 1", 1, "Turno 2", 2, "Turno 3", 3, 0)},
                      ClearCollect(
                          colHistEvMandril,
                          Filter(colHistMandrilDia, _turno = 0 || turno = _turno)
                      )
                  );

                  ClearCollect(
                      colHistFichasMandril,
                      ForAll(
                          Distinct(colHistEvMandril, origem_ID) As d,
                          With(
                              {f: LookUp(ficha_producao, ID = d.Value)},
                              {fID: d.Value, fOP: f.OP, fCod: f.cod_mandril, fLote: f.lote_mandril}
                          )
                      )
                  );

                  ClearCollect(
                      colHistMandril,
                      AddColumns(
                          colHistEvMandril As ev,
                          hMomento, ev.data_hora,
                          hTipo, ev.tipo.Value,
                          hMetros, ev.metros,
                          hSobra, Coalesce(ev.sobra, 0),
                          hCarreteis, ev.carretel,
                          hBitola, ev.bitola,
                          hOperador, ev.operador,
                          hRecurso, ev.recurso,
                          hOP, LookUp(colHistFichasMandril, fID = ev.origem_ID).fOP,
                          hCod, LookUp(colHistFichasMandril, fID = ev.origem_ID).fCod,
                          hLote, LookUp(colHistFichasMandril, fID = ev.origem_ID).fLote,
                          hDelta, If(
                              ev.tipo.Value <> "Fim",
                              Blank(),
                              With(
                                  {_ini: LookUp(colHistMandrilDia, origem_ID = ev.origem_ID && tipo.Value = "Início").metros},
                                  If(
                                      IsBlank(_ini),
                                      Blank(),
                                      Coalesce(ev.metros, 0) + Coalesce(ev.sobra, 0) - _ini
                                  )
                              )
                          )
                      )
                  ),
                  Notify("Não consegui carregar o histórico. Confira a rede e toque em Atualizar.", NotificationType.Error, 8000)
              );
              UpdateContext({ctxHistCarregando: false})
