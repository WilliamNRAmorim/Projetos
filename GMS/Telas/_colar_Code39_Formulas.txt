//  Code 39 (MEL-239) — o bipador da logística lê Code 39 (é o do BD Packing List antigo, `*GH781-8*`).
//  Aceita 0-9 A-Z - . espaço $ / + % ; o hífen do part number passa. 9 elementos por caractere
//  (barra/espaço alternados), `w` = largo (3 módulos). Validado com leitor ZXing em 100%, 83% e 60% da escala.
fx_C39Charset = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ-. *$/+%";

fx_C39Patterns = "nnnwwnwnnwnnwnnnnwnnwwnnnnwwnwwnnnnnnnnwwnnnwwnnwwnnnnnnwwwnnnnnnnwnnwnwwnnwnnwnnnnwwnnwnnwnnnnwnnwnnwnnwnnwwnwnnwnnnnnnnwwnnwwnnnwwnnnnnwnwwnnnnnnnnwwnwwnnnnwwnnnnwnnwwnnnnnnwwwnnwnnnnnnwwnnwnnnnwwwnwnnnnwnnnnnwnnwwwnnnwnnwnnnwnwnnwnnnnnnnwwwwnnnnnwwnnnwnnnwwnnnnnwnwwnwwnnnnnnwnwwnnnnnwwwwnnnnnnnwnnwnnnwwwnnwnnnnnwwnwnnnnnwnnnnwnwwwnnnnwnnnwwnnnwnnnwnnwnwnnnwnwnwnnnnwnwnnnwnnwnnnwnwnnnnwnwnwn";

fx_C39Pat(c: Text): Text =
    Mid(fx_C39Patterns, (Coalesce(Find(c, fx_C39Charset), 1) - 1) * 9 + 1, 9);

// Módulos 1/0 de "*TEXTO*": barra larga = 111, espaço largo = 000, 1 módulo de espaço entre caracteres.
fx_C39Bits(txt: Text): Text =
    Concat(
        ForAll(
            Sequence(Len(txt)) As _i,
            With(
                { _p: fx_C39Pat(Mid(txt, _i.Value, 1)) },
                Concat(
                    ForAll(
                        Sequence(9) As _j,
                        If(
                            Mod(_j.Value, 2) = 1,
                            If(Mid(_p, _j.Value, 1) = "w", "111", "1"),
                            If(Mid(_p, _j.Value, 1) = "w", "000", "0")
                        )
                    ),
                    Value
                ) & "0"
            )
        ),
        Value
    );

// SVG de 560 px de largura x `alt` de altura, igual ao tamanho do controle Image (sem escala fracionária).
// O módulo é inteiro (n px) e as barras ficam centralizadas com zona de silêncio.
fx_CodigoBarras39(txt: Text, alt: Number): Text =
    If(
        IsBlank(Trim(txt)),
        "",
        With(
            { _bits: fx_C39Bits("*" & Upper(Trim(txt)) & "*") },
            With(
                { _n: Max(1, RoundDown(520 / Len(_bits), 0)) },
                With(
                    { _x0: RoundDown((560 - Len(_bits) * _n) / 2, 0) },
                    "<svg xmlns='http://www.w3.org/2000/svg' width='560' height='" & Text(alt, "[$-en-US]0")
                    & "' viewBox='0 0 560 " & Text(alt, "[$-en-US]0") & "' shape-rendering='crispEdges'>"
                    & "<rect width='100%' height='100%' fill='white'/>"
                    & "<path fill='black' d='"
                    & Concat(
                        ForAll(
                            Sequence(Len(_bits)) As _i,
                            If(
                                Mid(_bits, _i.Value, 1) = "1",
                                "M" & Text(_x0 + (_i.Value - 1) * _n, "[$-en-US]0")
                                & " 6h" & Text(_n, "[$-en-US]0")
                                & "v" & Text(alt - 32, "[$-en-US]0")
                                & "h-" & Text(_n, "[$-en-US]0") & "z",
                                ""
                            )
                        ),
                        Value
                    )
                    & "'/>"
                    & "<text x='280' y='" & Text(alt - 8, "[$-en-US]0")
                    & "' text-anchor='middle' font-family='monospace' font-size='16' fill='black'>"
                    & fx_XmlSeguro(Upper(Trim(txt)))
                    & "</text></svg>"
                )
            )
        )
    );
