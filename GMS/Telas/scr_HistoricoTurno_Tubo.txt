                          With(
                              {f: LookUp(ficha_producao, ID = d.Value)},
                              {fID: d.Value, fOP: f.OP, fMangueira: f.Mangueira}
                          )
                      )
                  );

                  ClearCollect(
                      colHistTubo,
                      AddColumns(
                          colHistEv As ev,
                          hMomento, ev.data_hora,
                          hTipo, ev.tipo.Value,
                          hMetros, ev.metros,
                          hSobra, Coalesce(ev.sobra, 0),
                          hCarreteis, ev.carretel,
                          hBitola, ev.bitola,
                          hOperador, ev.operador,
                          hOP, LookUp(colHistFichas, fID = ev.origem_ID).fOP,
                          hMangueira, LookUp(colHistFichas, fID = ev.origem_ID).fMangueira,
                          hDelta, If(
                              ev.tipo.Value <> "Fim",
                              Blank(),
                              With(
                                  {_ini: LookUp(colHistTuboDia, origem_ID = ev.origem_ID && tipo.Value = "Início").metros},
                                  If(
                                      IsBlank(_ini),
                                      Blank(),
                                      Coalesce(ev.metros, 0) + Coalesce(ev.sobra, 0) - _ini
                                  )
                              )
                          )
                      )
                  ),
